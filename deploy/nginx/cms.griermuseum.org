# /etc/nginx/sites-available/cms.griermuseum.org
# Strapi CMS — admin panel + API + media uploads
# After certbot runs, it will add the SSL server block automatically.

server {
    listen 80;
    listen [::]:80;
    server_name cms.griermuseum.org;

    # Increase limits for media uploads
    client_max_body_size 50M;

    location / {
        proxy_pass         http://127.0.0.1:1337;
        proxy_http_version 1.1;
        proxy_set_header   Upgrade $http_upgrade;
        proxy_set_header   Connection 'upgrade';
        proxy_set_header   Host $host;
        proxy_set_header   X-Real-IP $remote_addr;
        proxy_set_header   X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header   X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;

        # Generous timeouts for large uploads
        proxy_read_timeout  300s;
        proxy_send_timeout  300s;
        proxy_connect_timeout 60s;
    }

    # Logs
    access_log /var/log/nginx/cms.griermuseum.org.access.log;
    error_log  /var/log/nginx/cms.griermuseum.org.error.log;
}
